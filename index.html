--!strict

local CombatService = {}

function CombatService:ResolveHit(
    attacker: Player,
    target: Humanoid,
    damage: number
)
    if not self:CanAttack(attacker) then
        return false
    end

    target:TakeDamage(damage)
    self.HitConfirmed:FireClient(attacker)
    return true
end

return CombatService
